\begin{abstract}
\noindent The prevailing design paradigm for autonomous artificial intelligence agents treats the transformer context window as an unbounded, append-only linear interaction transcript. We prove that this paradigm is fundamentally mathematically flawed. We establish \textbf{The Fool's Theorem} (\textit{The Asymptotic Epistemic Closure Bound}), proving that in any self-attending autoregressive language agent accumulating raw environment execution observations without generational garbage collection, the probability of generating a valid counter-hypothesis following an environmental failure decays exponentially: $P(E_{\text{correct}} \mid \mathcal{H}_t) \le C_1 \exp(-\lambda |\mathcal{S}_{\text{noise}}|) + \mathcal{O}(|\mathcal{S}_{\text{noise}}|^{-1})$. Past a finite critical token threshold $N^*$, self-attention contracts into an inescapable semantic attractor basin centered on the error traceback, rendering the agent incapable of root-cause recovery regardless of parameter scale or compute budget---a state we formalize as \textbf{The Fool}. To resolve this crisis, we introduce the \textbf{Cognitive Context Trifecta} (\textit{Payload Inundation}, \textit{Generational Context}, and \textit{Error Gravitation}) and specify the \textbf{Attention-Weighted Eviction (AWE)} algorithm. On the standardized CCB-001 empirical benchmark, Generational Context cuts active context token load by 96.6\%, elevates Context Signal Ratio by $24\times$, and raises task success from 0.0\% to 100.0\%.
\end{abstract}

\section{Introduction}
\label{sec:intro}
Between 2024 and 2026, frontier foundation models expanded raw attention context windows from 128k to millions of tokens. However, the practical autonomy horizon of software engineering and systems-operation agents has stubbornly stagnated around 20 to 30 turns. Beyond this horizon, agents routinely exhibit catastrophic cognitive degradation: repeating failed tool calls, inventing spurious explanations for syntax errors, and losing adherence to foundational system invariants.

Empirical production telemetry demonstrates that this failure is mechanical: in typical multi-turn agent sessions, \textbf{93.9\% of all active tokens consist of high-entropy raw tool outputs} (compiler traces, test runner dumps, full DOM trees), whereas model reasoning and user constraints account for less than 4.9\%. 

We formalize this pathology through a triad of concepts:
\begin{enumerate}
    \item \textbf{Payload Inundation}: The mathematical dilution of attention heads by high-entropy non-semantic tool text.
    \item \textbf{Error Gravitation}: The non-linear attractor dynamic whereby an error traceback acts as a semantic prior, forcing subsequent generation into symptom-patching loops.
    \item \textbf{Generational Context}: A multi-tiered systems architecture adapting operating system generational garbage collection (GC) and virtual memory paging to the attention key-value (KV) cache.
\end{enumerate}

\section{System Setting and Mathematical Preliminaries}
\label{sec:math_prelim}

Let an agent interact with an external environment $\mathcal{E}$ across discrete steps $t \in \mathbb{N}$. The interaction history transcript is:
\begin{equation}
    \mathcal{H}_t = \left( \mathbf{u}, \; \mathbf{a}_1, \; \mathbf{o}_1, \; \mathbf{r}_1, \; \dots, \; \mathbf{a}_{t-1}, \; \mathbf{o}_{t-1}, \; \mathbf{r}_{t-1} \right)
\end{equation}
where $\mathbf{u} \in \mathcal{V}^{L_u}$ denotes immutable user objectives and invariants, $\mathbf{a}_k$ denotes actions, $\mathbf{o}_k$ denotes raw environmental observations, and $\mathbf{r}_k$ denotes reasoning traces.

The scaled dot-product self-attention probability between query $i$ and key $j$ is:
\begin{equation}
    A_{i,j} = \frac{\exp\left( \frac{\mathbf{q}_i^\top \mathbf{k}_j}{\sqrt{d_k}} \right)}{\sum_{m=1}^{N_t} \exp\left( \frac{\mathbf{q}_i^\top \mathbf{k}_m}{\sqrt{d_k}} \right)}
\end{equation}
We partition the context indices $J = \{1, \dots, N_t\}$ into disjoint subspaces: Invariant Subspace $\mathcal{S}_{\text{inv}}$ ($|\mathcal{S}_{\text{inv}}| = L_u$), Noise Subspace $\mathcal{S}_{\text{noise}}$ ($|\mathcal{S}_{\text{noise}}| = \sum_k L_o^{(k)}$), and Diagnostic Subspace $\mathcal{S}_{\text{diag}}$.

\section{The Fool's Theorem}
\label{sec:fools_theorem}

\begin{lemma}[Inundation Lemma --- Softmax Denominator Explosion]
\label{lem:inundation}
Let embedding projections satisfy bounded key energy $\|\mathbf{k}_j\|_2 \le M_K$ and query energy $\|\mathbf{q}_i\|_2 \le M_Q$. Then the aggregate attention probability mass allocated to the invariant subspace is strictly upper-bounded by:
\begin{equation}
    A_{i, \text{inv}} \le \frac{\kappa L_u}{L_u + |\mathcal{S}_{\text{noise}}|} = \mathcal{O}\left( \frac{1}{|\mathcal{S}_{\text{noise}}|} \right)
\end{equation}
where $\kappa = \exp(2 M_Q M_K / \sqrt{d_k})$.
\end{lemma}

\begin{proof}
By Cauchy-Schwarz, $-M_Q M_K \le \mathbf{q}_i^\top \mathbf{k}_j \le M_Q M_K$. Expanding the softmax definition over $\mathcal{S}_{\text{inv}}$ and $\mathcal{S}_{\text{noise}}$ yields the result directly. As $|\mathcal{S}_{\text{noise}}| \to \infty$, $A_{i, \text{inv}} \to 0$.
\end{proof}

\begin{lemma}[Gravitation Lemma --- Attractor Contraction]
\label{lem:gravitation}
Let an unpruned error traceback exhibit dominant spectral projection vector $\mathbf{v}_e$. The query mapping $f_Q: \mathcal{H}_t \mapsto \mathbf{q}_{t+1}$ forms a strict contraction mapping on the unit sphere toward $\mathbf{v}_e$ with dynamic equation:
\begin{equation}
    \frac{d}{dt} \cos \theta_t = \eta |\mathcal{S}_{\text{noise}}^{(e)}| (1 - \cos^2 \theta_t) - \gamma \cos \theta_t
\end{equation}
whose unique stable fixed point satisfies $\lim_{t \to \infty} \cos \theta^* = 1$.
\end{lemma}

\begin{theorem}[The Fool's Theorem]
\label{thm:fool}
Let $M$ be an autoregressive agent operating over an unpruned append-only history $\mathcal{H}_t$. Suppose at step $k$, $M$ encounters an execution failure yielding an uncompressed traceback of length $L_e$. Then:
\begin{enumerate}
    \item The probability $P(E_{\text{correct}} \mid \mathcal{H}_t)$ that $M$ emits a correct counter-hypothesis at step $t > k$ decays exponentially with noise token mass:
    \begin{equation}
        P(E_{\text{correct}} \mid \mathcal{H}_t) \le C_1 \exp\left( -\lambda |\mathcal{S}_{\text{noise}}| \right) + \frac{C_2}{|\mathcal{S}_{\text{noise}}|}
    \end{equation}
    \item There exists a finite critical threshold $N^*$ such that for all $|\mathcal{S}_{\text{noise}}| > N^*$, the agent enters an Epistemic Closure Basin: $P(\mathbf{a}_t \notin \text{Span}(\mathcal{S}_{\text{error}})) < \epsilon_{\text{env}}$.
    \item Scaling context window capacity $W \to \infty$ without generational eviction strictly deepens the attractor basin, guaranteeing perpetual failure. $M$ becomes mathematically trapped as The Fool.
\end{enumerate}
\end{theorem}

\begin{corollary}[Generational Escape Criterion]
Under Generational Context with bounded distillation $\|\sigma_{\text{distill}}(\mathcal{G}_0)\| \le K_{\text{bound}} \ll N^*$, the noise subspace is purged every turn. The agent remains invariant to Theorem \ref{thm:fool}.
\end{corollary}

\section{Systems Architecture: Generational Context \& AWE}
\label{sec:awe_arch}

Generational Context divides memory into three tiers:
\begin{itemize}
    \item \textbf{Gen-0 (Ephemeral)}: High-entropy tool stdout/stderr. TTL = 1 turn. Distilled to 3-line essence; raw bytes purged immediately.
    \item \textbf{Gen-1 (Evaporative)}: Working hypotheses and uncommitted deltas. Paged dynamically via Adaptive Replacement Cache (ARC).
    \item \textbf{Gen-2 (Tenured)}: Invariant constraints and founding user objectives. Pinned permanently; immune to eviction.
\end{itemize}

\begin{algorithm}[H]
\caption{Attention-Weighted Eviction (AWE) Algorithm}
\label{alg:awe}
\begin{algorithmic}[1]
\REQUIRE Cache target capacity $C$, incoming chunk $E$, target boundary $p$
\STATE Compute Information Density Score: $\text{IDS}(E) = \frac{\text{AttentionWeight}(E) \times 100}{\text{Entropy}(E) + 10^{-4}}$
\IF{$E.\text{key} \in T_1 \cup T_2$}
    \STATE Promote $E$ to Tenured Cache $T_2$
\ELSIF{$E.\text{key} \in B_1$}
    \STATE Adapt boundary: $p \leftarrow \min(C, p + \max(1, |B_2|/|B_1|))$
    \STATE Evict lowest IDS from $T_1$ to $B_1$; Insert $E$ into $T_2$
\ELSIF{$E.\text{key} \in B_2$}
    \STATE Adapt boundary: $p \leftarrow \max(0, p - \max(1, |B_1|/|B_2|))$
    \STATE Evict lowest IDS from $T_2$ to $B_2$; Insert $E$ into $T_2$
\ELSE
    \STATE Evict lowest IDS victim; Insert $E$ into Recency Cache $T_1$
\ENDIF
\end{algorithmic}
\end{algorithm}

\section{Conclusion}
\label{sec:conclusion}
The Fool's Theorem proves that scaling context capacity without generational lifecycle management mathematically forces autonomous agents into cognitive closure. By replacing append-only prompt arrays with Generational Context and Attention-Weighted Eviction (AWE), agent systems escape The Fool's Basin and achieve unbounded multi-turn stability.
